cmake_minimum_required(VERSION 3.16)
project(moonlight-kindle VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_SKIP_RPATH TRUE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build static Moonlight core" FORCE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-2.0>=2.18)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec>=57.37 libavutil libswscale)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(UUID REQUIRED IMPORTED_TARGET uuid)
find_package(OpenSSL 1.1 REQUIRED)
find_package(EXPAT REQUIRED)
find_package(Threads REQUIRED)
set(UPSTREAM ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
# Fetch the pinned upstream core only when it is not already available locally.
if(NOT EXISTS "${UPSTREAM}/moonlight-common-c/CMakeLists.txt")
  include(FetchContent)
  FetchContent_Declare(moonlight_core
    GIT_REPOSITORY https://github.com/moonlight-stream/moonlight-common-c.git
    GIT_TAG b126e481a195fdc7152d211def17190e3434bcce
    GIT_SUBMODULES enet
    SOURCE_DIR "${UPSTREAM}/moonlight-common-c")
  FetchContent_GetProperties(moonlight_core)
  if(NOT moonlight_core_POPULATED)
    FetchContent_Populate(moonlight_core)
  endif()
endif()
add_subdirectory(${UPSTREAM}/moonlight-common-c common EXCLUDE_FROM_ALL)
add_library(gamestream STATIC
  ${UPSTREAM}/libgamestream/client.c ${UPSTREAM}/libgamestream/http.c
  ${UPSTREAM}/libgamestream/xml.c ${UPSTREAM}/libgamestream/mkcert.c)
target_include_directories(gamestream PUBLIC ${UPSTREAM}/libgamestream)
target_link_libraries(gamestream PUBLIC moonlight-common-c PkgConfig::CURL PkgConfig::UUID OpenSSL::SSL OpenSSL::Crypto EXPAT::EXPAT Threads::Threads)
target_compile_options(gamestream PRIVATE -Wno-deprecated-declarations)
add_library(kindle-video STATIC src/video.c src/geometry.c)
target_include_directories(kindle-video PUBLIC src)
target_link_libraries(kindle-video PUBLIC PkgConfig::FFMPEG moonlight-common-c Threads::Threads)
add_executable(moonlight-kindle src/main.c src/eink.c)
target_include_directories(moonlight-kindle PRIVATE vendor/fbink)
target_compile_options(moonlight-kindle PRIVATE -Wall -Wextra -Wno-deprecated-declarations -Wno-unused-parameter)
target_link_libraries(moonlight-kindle PRIVATE kindle-video gamestream PkgConfig::GTK ${CMAKE_DL_LIBS})
install(TARGETS moonlight-kindle RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(video-test tests/video_test.c)
  target_link_libraries(video-test PRIVATE kindle-video)
  add_test(NAME geometry COMMAND video-test)
endif()
if(BUILD_TESTING)
  add_executable(protocol-test tests/protocol_test.c)
  target_link_libraries(protocol-test PRIVATE gamestream)
endif()

if(BUILD_TESTING)
  add_executable(pair-test tests/pair_test.c)
  target_link_libraries(pair-test PRIVATE gamestream)
endif()

if(BUILD_TESTING)
  add_executable(ui-test EXCLUDE_FROM_ALL tests/ui_test.c src/eink.c)
  target_include_directories(ui-test PRIVATE vendor/fbink)
  target_compile_options(ui-test PRIVATE -UNDEBUG -Wno-deprecated-declarations)
  target_link_libraries(ui-test PRIVATE kindle-video gamestream PkgConfig::GTK ${CMAKE_DL_LIBS})
endif()
